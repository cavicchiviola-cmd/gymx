"#34c759" : "#0a84ff"); // Troviamo l'indice reale di questa specifica sessione nel database globale let realIndex = historyData.indexOf(sess); let exHtml = sess.data.map(ex => { let gruppo = ottieniGruppoMuscolare(ex.name); let setsHtml = ex.sets.map((s, i) => { if (gruppo === "Cardio") return `${s.reps} min ${parseFloat(s.kg) > 0 ? '| ' + s.kg + ' km' : ''}`; else { let prefs = s.type === 'warmup' ? 'W' : (s.type === 'drop' ? Note: ${ex.notes}
` : ''; return `
${ex.name}
${setsHtml}${noteHtml}
`; }).join(''); bodyHtml += `

${sess.routine}

Esercizi totali: ${sess.data.length}

${exHtml}
`; }); document.getElementById('modalTitle').innerText = `Dati del ${dataStr}`; document.getElementById('modalBody').innerHTML = bodyHtml; document.getElementById('dataModal').classList.add('active'); } // NUOVA FUNZIONE PER ELIMINARE LA SESSIONE function eliminaSessioneStorico(indexDatabase) { if(confirm("⚠️ Vuoi davvero eliminare questo intero allenamento dallo storico?\n\nI dati sui volumi e i massimali di quella giornata verranno rimossi.
